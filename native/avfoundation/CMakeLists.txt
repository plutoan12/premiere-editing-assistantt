cmake_minimum_required(VERSION 3.22)
if(NOT CMAKE_HOST_APPLE)
  message(FATAL_ERROR "AVFoundation targets require a real macOS SDK; Linux is not a native acceptance environment")
endif()
project(pea_avfoundation LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
add_library(pea_media STATIC Media.mm)
set_target_properties(pea_media PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(pea_media PRIVATE -fobjc-arc -Wno-deprecated-declarations)
target_link_libraries(pea_media PUBLIC "-framework Foundation" "-framework AVFoundation" "-framework CoreMedia" "-framework CoreGraphics" "-framework AudioToolbox")
add_executable(pea-avfoundation cli.cpp)
target_link_libraries(pea-avfoundation PRIVATE pea_media)
set(PEA_UXP_SDK "" CACHE PATH "User-provided official Adobe UXP Hybrid SDK root")
if(PEA_UXP_SDK)
  if(NOT EXISTS "${PEA_UXP_SDK}/src/api/UxpAddon.h")
    message(FATAL_ERROR "Official SDK src/api/UxpAddon.h not found. Do not substitute guessed/stub headers.")
  endif()
  add_library(pea-media MODULE addon.cpp)
  target_include_directories(pea-media PRIVATE "${PEA_UXP_SDK}/src/api")
  target_link_libraries(pea-media PRIVATE pea_media)
  set_target_properties(pea-media PROPERTIES PREFIX "" SUFFIX ".uxpaddon")
endif()
